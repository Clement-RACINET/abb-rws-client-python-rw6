    async def get_an_io_signal(
    client: RWSClient,
    network: str,
    unit: str,
    signal: str,
) -> httpx.Response:
    """
    Get an IO Signal.

    Route: ``GET /rw/iosystem/signals/{network}/{unit}/{signal}``
    Contraintes ABB: Not supported in bootserver mode. /rw/iosystem/signals/{network}/{unit}/{signal};state will return only the IO signal value.

    Args:
        client: Instance RWSClient ouverte.
        network: Paramètre de chemin URL.
        unit: Paramètre de chemin URL.
        signal: Paramètre de chemin URL.

    Returns:
        Réponse HTTP brute. Succès attendu : HTTP 200.

    Raises:
        RWSAuthenticationError: Sur HTTP 401.
        RWSNotFoundError: Sur HTTP 404.
        RWSHTTPError: Sur tout autre HTTP >= 400.
        # Codes ABB: BAD REQUEST(400)

    Example:
        # curl --digest -u "Default User":robotics "http://localhost/rw/iosystem/signals/Local/DRV_1/DRV1K1"
    """
    return await client.get(f"/rw/iosystem/signals/{network}/{unit}/{signal}")
